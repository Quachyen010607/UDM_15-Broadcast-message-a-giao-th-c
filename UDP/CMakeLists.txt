cmake_minimum_required(VERSION 3.16)
project(UdpQtModule LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core Network Widgets)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core Network Widgets)

add_library(udp_service src/UdpService.cpp src/UdpService.h)
target_include_directories(udp_service PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(udp_service PUBLIC Qt${QT_VERSION_MAJOR}::Core Qt${QT_VERSION_MAJOR}::Network)

add_executable(udp_demo demo/main.cpp)
target_link_libraries(udp_demo PRIVATE udp_service Qt${QT_VERSION_MAJOR}::Widgets)

include(CTest)
if(BUILD_TESTING)
    find_package(Qt${QT_VERSION_MAJOR} QUIET COMPONENTS Test)
    if(TARGET Qt${QT_VERSION_MAJOR}::Test)
        add_executable(test_udp tests/test_udp.cpp)
        target_link_libraries(test_udp PRIVATE udp_service Qt${QT_VERSION_MAJOR}::Test)
        add_test(NAME test_udp COMMAND test_udp)
    else()
        message(STATUS "Qt Test chưa được cài đặt: bỏ qua test_udp")
    endif()
endif()
