cmake_minimum_required(VERSION 3.16)
project(NetworkUI LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTOUIC_SEARCH_PATHS "${CMAKE_CURRENT_SOURCE_DIR}/ui")
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets Network)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets Network)
find_package(Qt${QT_VERSION_MAJOR} QUIET COMPONENTS WebSockets Test)
add_library(network_core
    src/Widgets.cpp src/Widgets.h src/UdpService.cpp src/UdpService.h src/Protocol.h)
target_include_directories(network_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_link_libraries(network_core PUBLIC Qt${QT_VERSION_MAJOR}::Widgets Qt${QT_VERSION_MAJOR}::Network)
if(TARGET Qt${QT_VERSION_MAJOR}::WebSockets)
    target_compile_definitions(network_core PUBLIC HAVE_WEBSOCKETS)
    target_link_libraries(network_core PUBLIC Qt${QT_VERSION_MAJOR}::WebSockets)
    message(STATUS "Qt WebSockets: enabled")
else()
    message(STATUS "Qt WebSockets: unavailable; TCP/UDP still work")
endif()
add_executable(network_server src/main_server.cpp src/ServerWindow.cpp src/ServerWindow.h ui/ServerWindow.ui)
target_link_libraries(network_server PRIVATE network_core)
add_executable(network_client src/main_client.cpp src/ClientWindow.cpp src/ClientWindow.h ui/ClientWindow.ui)
target_link_libraries(network_client PRIVATE network_core)
include(CTest)
if(BUILD_TESTING AND TARGET Qt${QT_VERSION_MAJOR}::Test)
    add_executable(test_network_ui tests/test_network_ui.cpp)
    target_link_libraries(test_network_ui PRIVATE network_core Qt${QT_VERSION_MAJOR}::Test)
    add_test(NAME network_ui_tests COMMAND test_network_ui)
endif()
